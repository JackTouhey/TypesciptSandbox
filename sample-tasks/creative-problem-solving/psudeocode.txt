START
    SET csvFile = csvFilePath
    SET csvContents = READ csvFile
    SET patientsFromCsv = empty array
    INITIALIZE dbConnection

    FOR EACH line IN csvContents  
        SET currentPatient = empty object
        SET patientData = array of values from splitting line on ','

        FOR EACH value IN patientData  
            IF value is a valid patient property THEN
                Add the value as a field to currentPatient
        END LOOP

        Add currentPatient to patientsFromCsv
    END LOOP

    FOR EACH patient IN patientsFromCsv
        SET databasePatient = CALL retrievePatientFromDatabase(patient)

        IF databasePatient is empty THEN
            CALL insertPatient(patient)
        ELSE
            SET patientsMatch = CALL doPatientsMatch(patient, databasePatient)

            IF patientsMatch is false THEN
                CALL updatePatient
            END IF
        END IF
    END LOOP

    FUNCTION retrievePatientFromDatabase(patient) 
        INITIALIZE query
        IF patient has medicare number THEN
            query = "SELECT first_name, last_name, dob, gender, address, mobile, email, medicare, ihi FROM patients WHERE medicare = patient.medicare"
        ELSE
            query = "SELECT first_name, last_name, dob, gender, address, mobile, email, medicare, ihi FROM patients WHERE ihi = patient.ihi"
        END IF

        RETURN EXECUTE query ON dbConnection
    END FUNCTION

    FUNCTION insertPatient(patient)
        SET query = "INSERT INTO patients ("
        INITIALIZE values AS empty array

        FOR EACH field OF patient
            IF field has a value THEN
                Append the fields name to query
                Add the fields value to values
            END IF
        END LOOP

        EXECUTE query, values ON dbConnection
    END FUNCTION
